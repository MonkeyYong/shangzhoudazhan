```js
const KNIGHT_OFFSETS = [[2,1],[2,-1],[-2,1],[-2,-1],[1,2],[1,-2],[-1,2],[-1,-2]];
function knightReach(piece) {
  const moves = [];
  for (const [dc, dr] of KNIGHT_OFFSETS) {
    const nc = piece.col + dc, nr = piece.row + dr;
    if (nc < 0 || nc > 18 || nr < 0 || nr > 18) continue;
    const occ = pieceAt(nc, nr);
    if (!occ) moves.push({ col: nc, row: nr, capture: false });
    else if (occ.side !== piece.side) moves.push({ col: nc, row: nr, capture: true });
    // 己方或无敌王：丢弃（不受阻挡、不吃无敌王）
  }
  return moves;
}
```

```js
  const items = layout.map(normLayoutItem).filter(Boolean);
  if (items.length === 0) return null;
  // 必须有双方各 1 king
  const hasWK = items.some(p => p[0]==="white" && p[1]==="king");
  const hasBK = items.some(p => p[0]==="black" && p[1]==="king");
  if (!hasWK || !hasBK) return null;
  // K10 对称
  for (const [side, type, c, r] of items) {
    const opp = side==="white" ? "black" : "white";
    const found = items.find(p => p[0]===opp && p[1]===type && p[2]===18-c && p[3]===18-r);
    if (!found) return null;
  }
  return items;
}
```
